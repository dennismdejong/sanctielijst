# IDP Authenticatie (Entra ID + OIDC) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zet Microsoft Entra ID vóór de Compliance Zoeker-app zodat de volledige app (web-UI én `/api/*`) alleen toegankelijk is voor geautoriseerde gebruikers, met MFA (via Conditional Access), audit-logging van zoek- en refresh-acties en RBAC (rollen `Compliance.Viewer`/`Compliance.Admin`).

**Architecture:** De FastAPI-app wordt zelf de OIDC-resource server: ze spreekt het Authorization Code flow with PKCE protocol met de bestaande Entra-tenant, valideert bij elke request het JWT (id-token in httponly cookie voor browsers, bearer access-token voor scripts) tegen de Entra-JWKS en haalt daaruit identiteit en rollen. Nieuwe modules `app/auth.py` (OIDC-client) en `app/audit.py` (SQLite-audit-log) worden aan `app/main.py` gekoppeld via FastAPI-dependencies. Een zelf-gehoste broker (Authentik/Keycloak) is bewust niet gekozen — alleen een `issuer`-configwijziging zou dat later mogelijk maken.

**Tech Stack:** Python 3.11, FastAPI, `PyJWT==2.10.1` (+ transitief `cryptography` voor RS256), `requests` (bestaand), stdlib `sqlite3`, `secrets`, `base64`, `hashlib`; pytest + FastAPI `TestClient`.

## Global Constraints

- Python 3.11+; nieuwe dependency alleen `PyJWT==2.10.1` in `requirements.txt`.
- Alle secret/credential-bewerkingen enkel via env (`AZURE_TENANT_ID`, `AZURE_CLIENT_ID`, `AZURE_REDIRECT_URI`, `AZURE_CLIENT_SECRET`, `AUTH_REQUIRED`, `COOKIE_SECURE`, `AUDIT_DB`). Niets hardcoden; niets committen.
- Rollen: `Compliance.Viewer` (mag zoeken), `Compliance.Admin` (mag zoeken + `POST /api/refresh`). Geen rollen in het token → fail-closed (`403`).
- Sessie-cookie: naam `sid`, `httponly=True`, `secure=<COOKIE_SECURE>`, `samesite="lax"`, `max_age` = token `exp`.
- Publiek blijft alleen `GET /api/health`. `/`, `/api/status`, `/api/me`, `/api/search`, `/api/refresh` worden auth-beschermd.
- Audit-log: SQLite `data/audit.sqlite` (pad via `AUDIT_DB`); schrijft query/filters, **geen zoekresultaten**; best-effort (een audit-fout blokkeert de request nooit).
- Auth aan/uit: auth is actief als OIDC-config (AZURE_*) aanwezig is én `AUTH_REQUIRED` niet `0`. Zonder config (bestaande tests, lokale dev) blijft de app ongewijzigd werken.
- UI-taal Nederlands; identifiers Engels. Geen code-commentaar tenzij niet-voor-de-hand liggend.
- Testsuite: `.venv/bin/python -m pytest -v`.

```python
import base64
import hashlib
import logging
import os
import secrets
from dataclasses import dataclass
from urllib.parse import urlencode

import jwt
import requests

logger = logging.getLogger(__name__)

COOKIE_NAME = "sid"
STATE_COOKIE = "oidc_state"
DEFAULT_SCOPE = "openid profile email"
CONFIG_TIMEOUT = 10


@dataclass
class OIDCConfig:
    issuer: str
    client_id: str
    redirect_uri: str
    scope: str = DEFAULT_SCOPE
    client_secret: str | None = None
    audience: str | None = None


def default_config() -> OIDCConfig | None:
    tenant = os.environ.get("AZURE_TENANT_ID")
    client_id = os.environ.get("AZURE_CLIENT_ID")
    redirect_uri = os.environ.get("AZURE_REDIRECT_URI")
    if not (tenant and client_id and redirect_uri):
        return None
    return OIDCConfig(
        issuer=f"https://login.microsoftonline.com/{tenant}/v2.0",
        client_id=client_id,
        redirect_uri=redirect_uri,
        client_secret=os.environ.get("AZURE_CLIENT_SECRET"),
        audience=os.environ.get("AZURE_AUDIENCE", client_id),
    )


def new_state() -> str:
    return secrets.token_urlsafe(32)


def new_verifier() -> str:
    return secrets.token_urlsafe(43)


def code_challenge(verifier: str) -> str:
    digest = hashlib.sha256(verifier.encode()).digest()
    return base64.urlsafe_b64encode(digest).rstrip(b"=").decode()


class OIDCClient:
    def __init__(self, config: OIDCConfig, session: requests.Session | None = None):
        self.config = config
        self._session = session or requests.Session()
        self._metadata: dict | None = None
        self._jwks: list[dict] | None = None

    def _discover(self) -> dict:
        if self._metadata is None:
            resp = self._session.get(f"{self.config.issuer}/.well-known/openid-configuration", timeout=CONFIG_TIMEOUT)
            resp.raise_for_status()
            self._metadata = resp.json()
        return self._metadata

    def _keys(self) -> list[dict]:
        if self._jwks is None:
            resp = self._session.get(self._discover()["jwks_uri"], timeout=CONFIG_TIMEOUT)
            resp.raise_for_status()
            self._jwks = resp.json().get("keys", [])
        return self._jwks

    def auth_url(self, state: str, verifier: str, nonce: str) -> str:
        params = {
            "client_id": self.config.client_id,
            "response_type": "code",
            "redirect_uri": self.config.redirect_uri,
            "response_mode": "query",
            "scope": self.config.scope,
            "state": state,
            "nonce": nonce,
            "code_challenge": code_challenge(verifier),
            "code_challenge_method": "S256",
        }
        return self._discover()["authorization_endpoint"] + "?" + urlencode(params)

    def exchange_code(self, code: str, verifier: str, nonce: str) -> str:
        data = {
            "grant_type": "authorization_code",
            "code": code,
            "redirect_uri": self.config.redirect_uri,
            "client_id": self.config.client_id,
            "code_verifier": verifier,
        }
        if self.config.client_secret:
            data["client_secret"] = self.config.client_secret
        resp = self._session.post(self._discover()["token_endpoint"], data=data, timeout=CONFIG_TIMEOUT)
        resp.raise_for_status()
        id_token = resp.json()["id_token"]
        claims = self.verify_token(id_token)
        if claims.get("nonce") != nonce:
            raise ValueError("nonce-mismatch")
        return id_token

    def verify_token(self, token: str, expected_aud: str | None = None) -> dict:
        aud = expected_aud or self.config.audience
        last_error = None
        for key in self._keys():
            try:
                return jwt.decode(
                    token,
                    key,
                    algorithms=["RS256"],
                    audience=aud,
                    issuer=self.config.issuer,
                    options={"verify_exp": True},
                )
            except jwt.InvalidTokenError as exc:
                last_error = exc
        raise ValueError(f"Token niet te verifiëren: {last_error}")

    def end_session_url(self, post_logout_redirect_uri: str) -> str:
        params = {
            "post_logout_redirect_uri": post_logout_redirect_uri,
            "client_id": self.config.client_id,
        }
        return self._discover()["end_session_endpoint"] + "?" + urlencode(params)
```

`app/audit.py`:
```python
import json
import logging
import sqlite3
from datetime import datetime, timezone
from pathlib import Path

logger = logging.getLogger(__name__)

SCHEMA = """
CREATE TABLE IF NOT EXISTS audit_log (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  ts TEXT NOT NULL,
  actor TEXT NOT NULL,
  action TEXT NOT NULL,
  detail TEXT NOT NULL,
  remote_ip TEXT,
  user_agent TEXT
);
"""


def connect(db_path: Path) -> sqlite3.Connection:
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(str(db_path), check_same_thread=False)
    conn.execute(SCHEMA)
    conn.commit()
    return conn


def log(db_path: Path, actor: str, action: str, detail: dict | None = None, remote_ip: str | None = None, user_agent: str | None = None) -> None:
    try:
        conn = connect(db_path)
        try:
            conn.execute(
                "INSERT INTO audit_log (ts, actor, action, detail, remote_ip, user_agent) VALUES (?, ?, ?, ?, ?, ?)",
                (datetime.now(timezone.utc).isoformat(), actor or "anonymous", action, json.dumps(detail or {}, ensure_ascii=False), remote_ip, user_agent),
            )
            conn.commit()
        finally:
            conn.close()
    except Exception:
        logger.exception("Audit-log schrijven mislukt")


def recent(db_path: Path, limit: int = 10) -> list[dict]:
    conn = connect(db_path)
    try:
        rows = conn.execute(
            "SELECT ts, actor, action, detail, remote_ip, user_agent FROM audit_log ORDER BY id DESC LIMIT ?",
            (limit,),
        ).fetchall()
        return [
            {"ts": r[0], "actor": r[1], "action": r[2], "detail": json.loads(r[3]), "remote_ip": r[4], "user_agent": r[5]}
            for r in rows
        ]
    finally:
        conn.close()
```

```python
    def _audit(request: Request, action: str, actor: str | None = None, **detail) -> None:
        if not auth_on:
            return
        try:
            claims = getattr(request.state, "user", None) or {}
            resolved = actor or claims.get("email") or claims.get("preferred_username") or claims.get("sub") or "anonymous"
            audit.log(
                audit_path,
                resolved,
                action,
                detail,
                request.client.host if request.client else None,
                request.headers.get("user-agent"),
            )
        except Exception:
            logger.exception("Audit-log schrijven mislukt")

    def _claims_from_request(request: Request) -> dict | None:
        token = request.cookies.get(auth.COOKIE_NAME)
        if not token:
            return None
        try:
            return oidc.verify_token(token)
        except Exception:
            logger.exception("Sessie-token ongeldig")
            return None

    def require_auth(request: Request) -> dict | None:
        if not auth_on:
            return None
        claims = _claims_from_request(request)
        if claims is None:
            if request.url.path.startswith("/api/") or "application/json" in request.headers.get("accept", ""):
                raise HTTPException(status_code=401, detail="Niet ingelogd")
            return RedirectResponse(url="/auth/login", status_code=307)
        request.state.user = claims
        return claims

    def require_role(role: str):
        def dep(request: Request) -> dict | None:
            if not auth_on:
                return None
            claims = getattr(request.state, "user", None) or {}
            if role not in set(claims.get("roles") or []):
                _audit(request, "denied", required_role=role)
                raise HTTPException(status_code=403, detail="Onvoldoende rechten")
            return claims
        return dep
```

```python
    @app.get("/auth/login")
    def login(request: Request):
        if not auth_on:
            return RedirectResponse(url="/", status_code=302)
        state = auth.new_state()
        verifier = auth.new_verifier()
        nonce = auth.new_state()
        url = oidc.auth_url(state, verifier, nonce)
        resp = RedirectResponse(url=url, status_code=302)
        resp.set_cookie(
            auth.STATE_COOKIE,
            json.dumps({"state": state, "verifier": verifier, "nonce": nonce}),
            httponly=True,
            secure=cookie_secure,
            samesite="lax",
            max_age=600,
        )
        return resp

    @app.get("/auth/callback")
    def callback(request: Request, code: str = Query(None), state: str = Query(None), error: str = Query(None)):
        if not auth_on:
            return RedirectResponse(url="/", status_code=302)
        stored_raw = request.cookies.get(auth.STATE_COOKIE)
        if error:
            _audit(request, "login_failed", reason=f"provider_error:{error}")
            return RedirectResponse(url="/?auth_error=provider", status_code=302)
        if not stored_raw:
            return RedirectResponse(url="/auth/login", status_code=302)
        try:
            stored = json.loads(stored_raw)
        except (json.JSONDecodeError, TypeError):
            stored = {}
        if stored.get("state") != state or not code:
            _audit(request, "login_failed", reason="state_mismatch")
            return RedirectResponse(url="/auth/login", status_code=302)
        try:
            token = oidc.exchange_code(code, stored["verifier"], stored["nonce"])
        except Exception:
            logger.exception("Code-uitwisseling mislukt")
            _audit(request, "login_failed", reason="exchange_failed")
            return RedirectResponse(url="/?auth_error=exchange", status_code=302)
        claims = oidc.verify_token(token)
        resp = RedirectResponse(url="/", status_code=302)
        max_age = max(0, int(claims["exp"]) - int(time.time()))
        resp.set_cookie(auth.COOKIE_NAME, token, httponly=True, secure=cookie_secure, samesite="lax", max_age=max_age)
        resp.delete_cookie(auth.STATE_COOKIE, path="/")
        _audit(request, "login", actor=claims.get("email") or claims.get("sub"))
        return resp

    @app.post("/auth/logout")
    def logout(request: Request, user: dict | None = Depends(require_auth)):
        claims = getattr(request.state, "user", None)
        _audit(request, "logout", actor=claims.get("email") or claims.get("sub") if claims else "anonymous")
        resp = RedirectResponse(url=oidc.end_session_url(str(request.base_url)) if auth_on else "/", status_code=302)
        resp.delete_cookie(auth.COOKIE_NAME, path="/")
        resp.delete_cookie(auth.STATE_COOKIE, path="/")
        return resp

    @app.get("/api/me")
    def me(user: dict | None = Depends(require_auth)):

        return {
            "authenticated": True,
            "email": claims.get("email") or claims.get("preferred_username"),
            "name": claims.get("name"),
```
